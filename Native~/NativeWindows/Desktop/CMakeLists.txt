cmake_minimum_required(VERSION 3.22)
if(APPLE)
	set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
	set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures (universal binary by default)")
endif()
project(VisualPinballNativeWindows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
	set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(UNITY_PLUGIN_API "" CACHE PATH "Installed Unity Editor Data/PluginAPI directory")
option(VPNW_BUILD_TESTS "Build native window fixtures" ON)
option(VPNW_DEPLOY "Copy the built library into VisualPinball.Unity/Plugins" ON)
if(NOT EXISTS "${UNITY_PLUGIN_API}/IUnityInterface.h")
	message(FATAL_ERROR "Set UNITY_PLUGIN_API to the installed Unity PluginAPI directory.")
endif()
get_filename_component(VPNW_API_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(VPNW_PACKAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SOURCES WindowRegistry.cpp)
if(APPLE)
	enable_language(OBJCXX)
	list(APPEND SOURCES Mac/MetalWindows.mm)
	set(PLATFORM_LIBRARIES "-framework Cocoa" "-framework QuartzCore" "-framework Metal" "-framework ApplicationServices")
	set(VPNW_PLUGIN_DIRECTORY "${VPNW_PACKAGE_ROOT}/VisualPinball.Unity/Plugins/osx" CACHE PATH "Deployment directory")
elseif(UNIX)
	find_package(Vulkan REQUIRED)
	find_package(X11 REQUIRED)
	if(NOT X11_Xrender_FOUND)
		message(FATAL_ERROR "The X Rendering Extension development files (libxrender-dev) are required.")
	endif()
	list(APPEND SOURCES Linux/X11Windows.cpp Linux/VulkanWindows.cpp)
	set(PLATFORM_LIBRARIES Vulkan::Vulkan X11::X11 X11::Xrender dl pthread m)
	set(VPNW_PLUGIN_DIRECTORY "${VPNW_PACKAGE_ROOT}/VisualPinball.Unity/Plugins/linux-x64" CACHE PATH "Deployment directory")
else()
	message(FATAL_ERROR "This CMake project builds the macOS and Linux library; Windows uses Native~/NativeWindows/Windows/Build.ps1.")
endif()
add_library(VisualPinballNativeWindows SHARED ${SOURCES})
target_include_directories(VisualPinballNativeWindows PRIVATE "${UNITY_PLUGIN_API}" "${VPNW_API_DIRECTORY}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(VisualPinballNativeWindows PRIVATE ${PLATFORM_LIBRARIES})
target_compile_options(VisualPinballNativeWindows PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
if(APPLE)
	target_compile_options(VisualPinballNativeWindows PRIVATE -fobjc-arc)
	set_target_properties(VisualPinballNativeWindows PROPERTIES PREFIX "lib" SUFFIX ".dylib")
endif()
if(VPNW_DEPLOY)
	add_custom_command(TARGET VisualPinballNativeWindows POST_BUILD
		COMMAND "${CMAKE_COMMAND}" -E make_directory "${VPNW_PLUGIN_DIRECTORY}"
		COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:VisualPinballNativeWindows>" "${VPNW_PLUGIN_DIRECTORY}/"
		COMMENT "Deploying VisualPinballNativeWindows to ${VPNW_PLUGIN_DIRECTORY}")
endif()
if(VPNW_BUILD_TESTS)
	enable_testing()
	add_executable(NativeWindowTests DesktopTests.cpp)
	if(APPLE)
		set_source_files_properties(DesktopTests.cpp PROPERTIES LANGUAGE OBJCXX)
		target_compile_options(NativeWindowTests PRIVATE -fobjc-arc)
	endif()
	target_include_directories(NativeWindowTests PRIVATE "${UNITY_PLUGIN_API}" "${VPNW_API_DIRECTORY}" "${CMAKE_CURRENT_SOURCE_DIR}")
	target_link_libraries(NativeWindowTests PRIVATE VisualPinballNativeWindows ${PLATFORM_LIBRARIES})
	add_test(NAME NativeWindows COMMAND NativeWindowTests)
endif()
