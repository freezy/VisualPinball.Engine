cmake_minimum_required(VERSION 3.22)
project(VpeAudioRouting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_library(AudioPluginVpeRouting SHARED RoutingPlugin.cpp)
add_library(VpePcm SHARED PcmRenderer.cpp AudioProcessing.cpp PcmOutput.cpp)
if(WIN32)
	target_sources(VpePcm PRIVATE PcmRenderer.def)
	target_link_libraries(VpePcm PRIVATE ole32 avrt)
endif()
if(MSVC)
	# Keep package deployment independent of the developer machine's VC++ redistributable.
	set_property(TARGET VpePcm PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
	target_compile_options(VpePcm PRIVATE /W4 /WX /permissive-)
else()
	target_compile_options(VpePcm PRIVATE -Wall -Wextra -Werror)
endif()
target_include_directories(AudioPluginVpeRouting PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
	# Unity's public ABI deliberately uses an anonymous struct/union.
	target_compile_options(AudioPluginVpeRouting PRIVATE /W4 /WX /wd4201 /permissive-)
else()
	target_compile_options(AudioPluginVpeRouting PRIVATE -Wall -Wextra -Werror)
endif()
option(VPE_AUDIO_BUILD_TESTS "Build offline routing fixtures" ON)
if(VPE_AUDIO_BUILD_TESTS)
	enable_testing()
	add_executable(AudioRoutingTests RoutingTests.cpp RoutingPlugin.cpp)
	target_include_directories(AudioRoutingTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
	if(MSVC)
		target_compile_options(AudioRoutingTests PRIVATE /W4 /WX /wd4201 /permissive-)
	else()
		target_compile_options(AudioRoutingTests PRIVATE -Wall -Wextra -Werror)
	endif()
	add_test(NAME AudioRouting COMMAND AudioRoutingTests)
	add_executable(PcmRendererTests PcmRendererTests.cpp PcmRenderer.cpp AudioProcessing.cpp PcmOutput.cpp)
	find_package(Threads REQUIRED)
	target_link_libraries(PcmRendererTests PRIVATE Threads::Threads)
	if(WIN32)
		target_link_libraries(PcmRendererTests PRIVATE ole32 avrt)
	endif()
	if(MSVC)
		target_compile_options(PcmRendererTests PRIVATE /W4 /WX /permissive-)
	else()
		target_compile_options(PcmRendererTests PRIVATE -Wall -Wextra -Werror)
	endif()
	add_test(NAME PcmRenderer COMMAND PcmRendererTests)
	add_executable(AudioProcessingTests AudioProcessingTests.cpp PcmRenderer.cpp AudioProcessing.cpp)
	target_link_libraries(AudioProcessingTests PRIVATE Threads::Threads)
	if(MSVC)
		target_compile_options(AudioProcessingTests PRIVATE /W4 /WX /permissive-)
	else()
		target_compile_options(AudioProcessingTests PRIVATE -Wall -Wextra -Werror)
	endif()
	add_test(NAME AudioProcessing COMMAND AudioProcessingTests)
endif()

include_directories(SYSTEM ThirdParty/Signalsmith/basics/include ThirdParty/Signalsmith/dsp/include)
