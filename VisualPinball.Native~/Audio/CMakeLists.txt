cmake_minimum_required(VERSION 3.22)
project(VpeAudioRouting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_library(AudioPluginVpeRouting SHARED RoutingPlugin.cpp)
target_include_directories(AudioPluginVpeRouting PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
	# Unity's public ABI deliberately uses an anonymous struct/union.
	target_compile_options(AudioPluginVpeRouting PRIVATE /W4 /WX /wd4201 /permissive-)
else()
	target_compile_options(AudioPluginVpeRouting PRIVATE -Wall -Wextra -Werror)
endif()
option(VPE_AUDIO_BUILD_TESTS "Build offline routing fixtures" ON)
if(VPE_AUDIO_BUILD_TESTS)
	enable_testing()
	add_executable(AudioRoutingTests RoutingTests.cpp RoutingPlugin.cpp)
	target_include_directories(AudioRoutingTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
	if(MSVC)
		target_compile_options(AudioRoutingTests PRIVATE /W4 /WX /wd4201 /permissive-)
	else()
		target_compile_options(AudioRoutingTests PRIVATE -Wall -Wextra -Werror)
	endif()
	add_test(NAME AudioRouting COMMAND AudioRoutingTests)
endif()
